1.0.1 Variable transformations

Key variables from each file

  • Constructed files (ethiopia, india, peru, vietnam): child’s ID (childid), cohort (yc), round, cluster ID (clustid), age in years (ageyea), sex (chsex), child’s ethnic group (chethnic), child’s religious affiliation (chldrel), life satisfaction (cladder)

  • SDQ-E files (ethiopia_r3-5, india_r3-5, peru_r3-5, vietnam_r3-5): strengths and difficulties questionnaire emotional subscale - psychological distress (comprised of 5 items) collected at rounds 3, 4 and 5 for the 1994-cohort at ages 15, 19, and 22 respectively

## Country prefixes for childid
prefixes <- c(ethiopia = "ET", india = "IN", peru = "PE", vietnam = "VN")

## Fix childid: make sure there are 6 digits and add country prefix
fix_childid <- function(id, prefix) {
  id <- as.character(id)
  ifelse(str_detect(id, "^[A-Za-z]"),                 # if already full format
         id,                                          # leave it alone
         paste0(prefix, str_pad(id, 6, "left", "0"))) # otherwise reconstruct
}

## SDQ items come in two naming schemes across countries/rounds
scheme_headac <- function(r) c(
  headache = paste0("HEADACR", r),
  worry    = paste0("WRYLOTR", r),
  unhappy  = paste0("UNHPPYR", r),
  nervous  = paste0("NRVSITR", r),
  fears    = paste0("MNYFERR", r)
)
scheme_feay <- function(r) c(
  headache = paste0("FEAY13R", r),
  worry    = paste0("FEAY05R", r),
  unhappy  = paste0("FEAY20R", r),
  nervous  = paste0("FEAY38R", r),
  fears    = paste0("FEAY28R", r)
)

## Which scheme each country uses at R4/R5 (R3 is same across countries)
country_scheme <- list(
  india    = scheme_feay,
  peru     = scheme_headac,
  ethiopia = scheme_feay,
  vietnam  = scheme_feay
)

## Clean SDQ files: select + rename items and ID, then fix childid
sdq <- imap(sdq_list, function(df, nm) {
  ctry <- str_extract(nm, "^[a-z]+")
  rnum <- str_extract(nm, "[3-5]")
  pfx  <- prefixes[[ctry]]
  
  df <- df |> rename_with(toupper)
  
  # R3 uses HEADAC everywhere; R4/R5 use the country's scheme
  items <- if (rnum == "3") scheme_headac(rnum) else country_scheme[[ctry]](rnum)
  items <- setNames(items, paste0(names(items), "_r", rnum))
  
  idc <- if (rnum == "3") "CHILDID" else "CHILDCODE"
  
  df |>
    select(all_of(c(childid = idc, items))) |>
    mutate(childid = fix_childid(childid, pfx))
})

SDQ-E needs to be rescaled and standardised as measurement was different across rounds and countries.

  • Round 3: 1 - not true for you, 2 - a little true for you, 3 - certainly true for you, 77 - NK, 88 - N/A, 99 - Missing

  • Rounds 4 and 5: 1 - strongly disagree, 2 - disagree, 3 - agree, 4 - strongly agree, 77 - NK, 79 - refused to answer

SDQ sub-scale is normally 0 - not true, 1 - somewhat true, 2 - certainly true -> 10 full score.

Therefore, scale of 1-3 (round 3) will be rescaled to 0-2. As for scale of 1-4 (rounds 4-5), 1,2 will be collapsed to 0 (not true), 3>1, and 4->2.

The full score will be calculated as the mean of non-missing data * 5.

Rescale SDQ

# Rescale function; unrecognised values (77/79/88/99) fall all become NA
rescale_3pt <- function(x) case_when(x == 1 ~ 0, x == 2 ~ 1, x == 3 ~ 2, TRUE ~ NA_real_)
rescale_4pt <- function(x) case_when(x %in% c(1, 2) ~ 0, x == 3 ~ 1, x == 4 ~ 2, TRUE ~ NA_real_)

# Format at R4/R5 (R3 is 3pt for everyone) — all four confirmed against the data
format_r45 <- c(india = "4pt", peru = "3pt", ethiopia = "4pt", vietnam = "4pt")

sdq <- imap(sdq, function(df, nm) {
  ctry <- str_extract(nm, "^[a-z]+")
  rnum <- str_extract(nm, "[3-5]")
  cols <- paste0(c("headache", "worry", "unhappy", "nervous", "fears"), "_r", rnum)
  
  fmt     <- if (rnum == "3") "3pt" else format_r45[[ctry]]
  rescale <- if (fmt == "3pt") rescale_3pt else rescale_4pt
  
  df |>
    mutate(across(all_of(cols), ~ rescale(as.numeric(.x)))) |>
    mutate(
      nonmiss = rowSums(!is.na(across(all_of(cols)))),
      !!paste0("emo_r", rnum) := if_else(
        nonmiss > 2, # need to have answered 3 or more out of 5 items to be included
        round(rowMeans(across(all_of(cols)), na.rm = TRUE) * 5),
        NA_real_
      )
    ) |>
    select(-nonmiss)
})

1.0.2.1 Create variable ‘equivalent age’

Although we have age in years as the variable for real age, we need an equivalent age of each round i.e. in round 3 for example, the 2001-cohort is aged 8 and the 1994-cohort is aged 15.

Code 
age_yc <- c("1"=1, "2"=5, "3"=8, "4"=12, "5"=15, "6"=19, "7"=22)
age_oc <- c("1"=8, "2"=12, "3"=15, "4"=19, "5"=22, "6"=26, "7"=29)

add_age <- function(df) {
  df %>%
    mutate(equivalent_age = if_else(yc == 1, age_yc[as.character(round)],
                                    age_oc[as.character(round)])) %>%
    filter(!is.na(equivalent_age)) %>%
    group_by(childid, equivalent_age) %>%
    slice_max(!is.na(cladder), n = 1, with_ties = FALSE) %>% # In round 6 there are multiple rows for phone calls but we only keep the row that has cladder data
    ungroup()
}

stacked <- map(merged_data, add_age)

# Specify majority religions
majority_religion <- list(ethiopia = "orthodox", india = "hindu", peru = "catholic", vietnam = c("buddhist", "none", "ancestor worship"))

classify <- function(df, ctry) {
  maj <- majority_religion[[tolower(ctry)]]
  df |> 
    mutate(
      chldrel = as.character(as_factor(chldrel)),
     minority = case_when(
       is.na(chldrel) ~ NA_character_,
       tolower(chldrel) %in% maj ~ "Majority",
       TRUE ~ "Minority"
     )
      )
}

# Mean cladder by age, overall and by group, per country
plot_trajectory <- function(df, ctry) {
  # split version
  df |>
    filter(!is.na(cladder), !is.na(equivalent_age)) |>
    group_by(equivalent_age) |>
    summarise(mean_cl = mean(cladder), n = n(), .groups = "drop") |>
    ggplot(aes(equivalent_age, mean_cl)) +
    geom_line() + geom_point() +
    scale_x_continuous(breaks = c(8, 12, 15, 19, 22, 26, 29)) +
    scale_y_continuous(limits = c(4, 7)) +
    labs(title = ctry, x = "Age", y = "Mean life satisfaction") +
    theme_minimal()
}

# Mean cladder by age, overall and by group, per country
plot_trajectory_minmaj <- function(df, ctry) {
  # split version
  df |>
    filter(!is.na(cladder), !is.na(equivalent_age)) |>
    group_by(equivalent_age, minority) |>
    summarise(mean_cl = mean(cladder),
              se = sd(cladder)/sqrt(n()),
              n = n(), .groups = "drop") |>
    ggplot(aes(equivalent_age, mean_cl, colour = minority)) +
    geom_line() + geom_point() +
    scale_x_continuous(breaks = c(8, 12, 15, 19, 22, 26, 29)) +
    scale_y_continuous(limits = c(4, 7)) +
    labs(title = ctry, x = "Age", y = "Mean life satisfaction") +
    geom_ribbon(aes(ymin = mean_cl - 1.96*se, ymax = mean_cl + 1.96*se, fill = minority), alpha = 0.2, colour = NA) +
    scale_colour_manual(values = group_cols) +
    scale_fill_manual(values = group_cols) +
    theme_minimal()
}

Welch’s t test is preferred since the minority and majority groups do not have equal variances nor sample sizes. We can test whether the group means between minorities and majorities are statistically significant.

Note that these are purely descriptive, we’re doing many independent t-tests with no correction so this is more for establishing overall patterns.

diff_min_maj is the minority - majority, i.e. if diff_min_maj is positive for cladder, that means that minorities have higher life satisfaction; whereas if it’s positive for sdq, that means minorities have lower psychological distress (since the two scales are coded differently).

Summary

  • India: cladder at age 22: minority better life satisfaction than majority

# Build lavaan growth syntax from a vector of ages, centred at the first age.
# type: "intercept", "linear", or "conditional"
make_spec <- function(ages, type = "linear") {
  t <- ages - ages[1]                              # time scores, centred at first age
  cols <- paste0("cladder_age", ages)
  i <- paste0("i =~ ", paste0("1*", cols, collapse = " + "))
  if (type == "intercept") return(i)
  s <- paste0("s =~ ", paste0(t, "*", cols, collapse = " + "))
  spec <- paste(i, s, sep = "\n")
  if (type == "conditional")
    spec <- paste(spec, "i ~ minority + chsex", "s ~ minority + chsex", sep = "\n")
  spec
}

# Fit one model type for one cohort, across all countries
fit_all <- function(ages_list, type, cohort_val) {
  imap(wide, function(df, ctry) {
    spec <- make_spec(ages_list[[ctry]], type)
    d <- df |> filter(yc == cohort_val, !is.na(minority), !is.na(chsex)) |>
      mutate(minority = as.numeric(minority), chsex = as.numeric(chsex))
    tryCatch(growth(spec, data = d, missing = "fiml"), error = function(e) NULL)
  })
}

get_fit <- function(fits) {
  extract <- function(f) {
    if (is.null(f)) return(NULL)
    as_tibble(as.list(fitMeasures(f, c("cfi","tli","rmsea"))))
  }
  map_dfr(fits, extract, .id = "country")
}

get_min_effect <- function(fits) {
  extract <- function(f) {
    if (is.null(f)) return(NULL)
    parameterEstimates(f) |>
      filter(rhs == "minority", lhs %in% c("i","s")) |>
      select(lhs, est, se, pvalue)
  }
  map_dfr(fits, extract, .id = "country")
}

# Build implied minority & majority trajectories from a conditional fit
implied_traj <- function(fit, ages) {
  if (is.null(fit)) return(NULL)
  cf <- coef(fit)
  t <- ages - ages[1]
  # majority = reference (minority=0, sex held at 0); minority adds its coef
  maj_i <- cf["i~1"]; maj_s <- cf["s~1"]
  min_i <- maj_i + cf["i~minority"]; min_s <- maj_s + cf["s~minority"]
  tibble(
    age = rep(ages, 2),
    group = rep(c("Majority","Minority"), each = length(ages)),
    fit = c(maj_i + maj_s*t, min_i + min_s*t)
  )
}
